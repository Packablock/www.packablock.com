document.addEventListener("DOMContentLoaded", function() {
  const params = new URLSearchParams(window.location.search);
  const checkoutBtn = document.getElementById("stripe-checkout-btn");
  if (checkoutBtn && params.toString()) {
    const originalHref = checkoutBtn.getAttribute("href");
    const separator = originalHref.includes("?") ? "&" : "?";
    // Forward client_reference_id or custom utm params
    const utmSource = params.get("utm_source") || "";
    const utmCampaign = params.get("utm_campaign") || "";
    const ref = utmSource ? `${utmSource}_${utmCampaign}` : "";
    
    if (ref) {
      checkoutBtn.setAttribute("href", `${originalHref}${separator}client_reference_id=${encodeURIComponent(ref)}`);
    }
  }
});
